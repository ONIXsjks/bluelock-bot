import telebot
from telebot import types
import os

BOT_TOKEN = os.environ.get('BOT_TOKEN')
bot = telebot.TeleBot(BOT_TOKEN)

MINI_APP_URL = "https://your-app.up.railway.app"

@bot.message_handler(commands=['start'])
def start(message):
    markup = types.ReplyKeyboardMarkup(resize_keyboard=True)
    btn = types.KeyboardButton("⚽ ورود به بلو لاک", web_app=types.WebAppInfo(url=MINI_APP_URL))
    markup.add(btn)
    bot.send_message(message.chat.id, "برای بازی روی دکمه زیر بزن:", reply_markup=markup)

bot.polling()
